# ==============================================================================
# BASIC SUPPRESSION MODE RULES
# ==============================================================================

## Disable Respawn on Death
Rule: "Disable Respawn"
Event: OnPlayerDied
Conditions: None
Actions:
  - DisableRespawn(EventPlayer)
  - SetSpectatorTarget(EventPlayer, RandomTeammate)

## Check Team Elimination
Rule: "Check Team Eliminated"
Event: OnPlayerDied
Conditions: CountAlive(EventPlayer.Team) == 0
Actions:
  - RoundVictory(OpposingTeam)
  - AddScoreToTeam(OpposingTeam, 1)

## Round Start Setup
Rule: "Round Start"
Event: OnRoundStart
Conditions: None
Actions:
  - EnableRespawnForAll()
  - SetRoundTimer(180)
  - EnableDamage()
  - ResetPlayerStats()

## Round Timer Expired
Rule: "Round Timeout"
Event: OnRoundTimerExpired
Conditions: None
Actions:
  - If CountAlive(Team1) > CountAlive(Team2)
      Then RoundVictory(Team1)
  - Else If CountAlive(Team2) > CountAlive(Team1)
      Then RoundVictory(Team2)
  - Else
      Then DrawRound()

## Match Victory Check
Rule: "Check Match Victory"
Event: OnRoundEnd
Conditions: None
Actions:
  - If Team1Score >= 6
      Then MatchVictory(Team1)
  - Else If Team2Score >= 6
      Then MatchVictory(Team2)
  - Else
      Then StartNewRound()

# ==============================================================================
# DEMOLITION MODE RULES (Bomb Plant/Defuse)
# ==============================================================================

## Variables Needed
GlobalVariables:
  - BombPlanted (Boolean)
  - BombCarrier (Player)
  - PlantTimer (Number)
  - DefuseTimer (Number)
  - PostPlantTimer (Number)

## Assign Bomb Carrier
Rule: "Assign Bomb Carrier"
Event: OnRoundStart
Conditions: None
Actions:
  - BombCarrier = RandomPlayer(Attackers)
  - SetPlayerVariable(BombCarrier, "HasBomb", True)
  - DisplayMessage(BombCarrier, "You have the bomb!")
  - MarkPlayerWithIcon(BombCarrier, BombIcon)

## Plant Bomb
Rule: "Plant Bomb"
Event: OnPlayerInteract
Conditions:
  - EventPlayer == BombCarrier
  - IsAtObjective(EventPlayer)
  - BombPlanted == False
Actions:
  - StartProgressBar(EventPlayer, 5)
  - OnProgressComplete:
      - BombPlanted = True
      - PostPlantTimer = 45
      - DisplayMessageToAll("Bomb has been planted!")
      - StartPostPlantTimer(45)
      - MarkObjectiveWithIcon(ObjectiveSite, BombIcon)

## Defuse Bomb
Rule: "Defuse Bomb"
Event: OnPlayerInteract
Conditions:
  - EventPlayer.Team == Defenders
  - IsAtBombSite(EventPlayer)
  - BombPlanted == True
Actions:
  - StartProgressBar(EventPlayer, 7)
  - OnProgressComplete:
      - BombPlanted = False
      - RoundVictory(Defenders)
      - DisplayMessageToAll("Bomb has been defused!")

## Bomb Explodes
Rule: "Bomb Detonates"
Event: OnTimerExpired
Conditions: PostPlantTimer <= 0 AND BombPlanted == True
Actions:
  - RoundVictory(Attackers)
  - DisplayMessageToAll("Bomb detonated!")
  - CreateExplosion(BombLocation)

## Bomb Carrier Killed
Rule: "Bomb Dropped"
Event: OnPlayerDied
Conditions: EventPlayer == BombCarrier
Actions:
  - DropBomb(EventPlayer.Position)
  - MarkLocationWithIcon(EventPlayer.Position, BombIcon)
  - DisplayMessageToAll("Bomb has been dropped!")

## Pick Up Dropped Bomb
Rule: "Pick Up Bomb"
Event: OnPlayerInteract
Conditions:
  - EventPlayer.Team == Attackers
  - IsNearDroppedBomb(EventPlayer)
Actions:
  - BombCarrier = EventPlayer
  - SetPlayerVariable(BombCarrier, "HasBomb", True)
  - DisplayMessage(EventPlayer, "You have the bomb!")
  - MarkPlayerWithIcon(BombCarrier, BombIcon)

# ==============================================================================
# EXTRACTION MODE RULES (Hostage Rescue)
# ==============================================================================

## Variables Needed
GlobalVariables:
  - HostageRescued (Boolean)
  - HostageCarrier (Player)
  - HostageLocation (Vector)
  - ExtractionZone (Vector)

## Rescue Hostage
Rule: "Rescue Hostage"
Event: OnPlayerInteract
Conditions:
  - EventPlayer.Team == Attackers
  - IsNearHostage(EventPlayer)
  - HostageRescued == False
Actions:
  - StartProgressBar(EventPlayer, 3)
  - OnProgressComplete:
      - HostageCarrier = EventPlayer
      - HostageRescued = True
      - DisplayMessage(EventPlayer, "Escorting hostage")
      - SetMovementSpeed(EventPlayer, 0.7)
      - AttachHostageToPlayer(EventPlayer)

## Extract Hostage
Rule: "Extract Hostage"
Event: OnPlayerEnterZone
Conditions:
  - EventPlayer == HostageCarrier
  - IsInExtractionZone(EventPlayer)
Actions:
  - RoundVictory(Attackers)
  - DisplayMessageToAll("Hostage extracted!")

## Hostage Carrier Killed
Rule: "Hostage Carrier Killed"
Event: OnPlayerDied
Conditions: EventPlayer == HostageCarrier
Actions:
  - HostageRescued = False
  - HostageLocation = EventPlayer.Position
  - MarkLocationWithIcon(HostageLocation, HostageIcon)
  - DisplayMessageToAll("Hostage needs to be rescued again!")

# FRIENDLY FIRE RULES (Optional)
# ==============================================================================

## Track Team Kills
Rule: "Team Kill Tracking"
Event: OnPlayerDied
Conditions: Attacker.Team == Victim.Team
Actions:
  - AddToPlayerVariable(Attacker, "TeamKills", 1)
  - DisplayMessage(Attacker, "WARNING: Team Kill!")
  - SubtractFromPlayerScore(Attacker, 100)

## Auto-Kick for Team Killing
Rule: "Team Kill Auto-Kick"
Event: OnPlayerDied
Conditions:
  - Attacker.Team == Victim.Team
  - GetPlayerVariable(Attacker, "TeamKills") >= 3
Actions:
  - DisplayMessageToAll("Player kicked for team killing")
  - KickPlayer(Attacker)

## Reflect Team Damage (Hardcore)
Rule: "Reflect Team Damage"
Event: OnPlayerDamaged
Conditions:
  - Attacker.Team == Victim.Team
  - Attacker != Victim
Actions:
  - CancelDamage()
  - ApplyDamage(Attacker, EventDamage)

## Spawn Protection
Rule: "Spawn Protection"
Event: OnPlayerSpawn
Conditions: None
Actions:
  - SetPlayerInvulnerable(EventPlayer, True)
  - SetPlayerInvisible(EventPlayer, True)

Conditions: None
